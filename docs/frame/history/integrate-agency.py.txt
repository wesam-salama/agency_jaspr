def links(copy):
    entries=[]
    destinations = ['site/index.html?lang=ar','demo-journey-ar.mp4','frame-source.zip'] if copy is COPY['ar'] else ['site/index.html','demo-journey.mp4','frame-source.zip']
    for label,href,fmt in zip(copy['links'],destinations,['HTML','MP4','ZIP']):
        entries.append(f"      CaseDownload(label: {q(label)}, href: {q(ASSETS+href)}, format: {q(fmt)}" + (", isDownload: false" if fmt!='ZIP' else '') + '),')
    return '    downloads: [\n'+'\n'.join(entries)+'\n    ],\n'

def common(c):
    fields=['tag','heroAlt','client','role','timeline','deliverables']
    s=''.join(f'    {key}: {q(c[key])},\n' for key in fields)
    s+='    lede: CaseLede('+', '.join(map(q,c['lede']))+'),\n'
    s+='    paragraphs: [\n'+''.join('      '+q(p)+',\n' for p in c['paragraphs'])+'    ],\n'
    return s

def replace_between(file,start,end,new):
    p=ROOT/file;t=p.read_text();a=t.index(start);b=t.index(end,a+len(start));p.write_text(t[:a]+new+t[b:])
